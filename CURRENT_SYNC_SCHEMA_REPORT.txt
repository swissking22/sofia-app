--- Migration 2: migrateStudentIdSystem(db) ---
function migrateStudentIdSystem(db: Database.Database): void {
  db.exec(`
    CREATE TABLE IF NOT EXISTS student_id_settings (
      id TEXT PRIMARY KEY NOT NULL,
      school_id TEXT NOT NULL UNIQUE,
      pattern TEXT NOT NULL DEFAULT '{SCHOOL}-{SEQ}',
      sequence_scope TEXT NOT NULL DEFAULT 'school',
      next_sequence_number INTEGER NOT NULL DEFAULT 1,
      created_at TEXT NOT NULL,
      updated_at TEXT NOT NULL,
      deleted_at TEXT,
      sync_status TEXT NOT NULL DEFAULT 'pending',
      version INTEGER NOT NULL DEFAULT 1,
      FOREIGN KEY (school_id) REFERENCES schools(id)
    );

    CREATE INDEX IF NOT EXISTS idx_student_id_settings_school ON student_id_settings(school_id);
  `)

  const studentCols = (
    db.prepare(`PRAGMA table_info(students)`).all() as Array<{ name: string }>
  ).map((c) => c.name)
  const additions = [
    { name: 'school_id', sql: `ALTER TABLE students ADD COLUMN school_id TEXT` },
    {
      name: 'student_code',
      sql: `ALTER TABLE students ADD COLUMN student_code TEXT`,
    },
    {
      name: 'admission_year',
      sql: `ALTER TABLE students ADD COLUMN admission_year INTEGER`,
    },
    {
      name: 'academic_level_id',
      sql: `ALTER TABLE students ADD COLUMN academic_level_id TEXT`,
    },
    {
      name: 'sequence_number',
      sql: `ALTER TABLE students ADD COLUMN sequence_number INTEGER`,
    },
  ]
  for (const col of additions) {
    if (!studentCols.includes(col.name)) db.exec(col.sql)
  }

  // Safe migration for existing student_id_settings:
  // 1. Migrate default pattern from {SCHOOL}-{YEAR}-{SEQ} to {SCHOOL}-{SEQ}
  // 2. Migrate sequence_scope from 'year' to 'school'
  // 3. Preserve all existing student records, IDs, and student codes unchanged
  db.exec(`
    UPDATE student_id_settings
    SET pattern = '{SCHOOL}-{SEQ}',
        updated_at = datetime('now'),
        version = version + 1,
        sync_status = 'pending'
    WHERE pattern = '{SCHOOL}-{YEAR}-{SEQ}';

    UPDATE student_id_settings
    SET sequence_scope = 'school',
        updated_at = datetime('now'),
        version = version + 1,
        sync_status = 'pending'
    WHERE sequence_scope = 'year';
  `)

  // Report existing students with missing student_code (do not silently regenerate)
  try {
    const missingCodes = db
      .prepare(
        `SELECT count(*) as count FROM students WHERE (student_code IS NULL OR TRIM(student_code) = '') AND deleted_at IS NULL`
      )
      .get() as { count: number } | undefined
    if (missingCodes && missingCodes.count > 0) {
      console.warn(
        `[sofia] Note: Found ${missingCodes.count} student(s) with missing student_code. Leaving existing records unchanged.`
      )
    }
  } catch {
    /* ignore */
  }

  // Advance next_sequence_number if existing students have higher sequence numbers
  try {
    const settingsList = db
      .prepare(
        `SELECT id, school_id, next_sequence_number FROM student_id_settings WHERE deleted_at IS NULL`
      )
      .all() as Array<{ id: string; school_id: string; next_sequence_number: number }>

    for (const setting of settingsList) {
      const maxRow = db
        .prepare(
          `SELECT MAX(sequence_number) as max_seq FROM students WHERE school_id = ? AND deleted_at IS NULL`
        )
        .get(setting.school_id) as { max_seq: number | null } | undefined

      if (maxRow?.max_seq != null && maxRow.max_seq >= setting.next_sequence_number) {
        db.prepare(
          `UPDATE student_id_settings
           SET next_sequence_number = ?,
               updated_at = datetime('now'),
               version = version + 1,
               sync_status = 'pending'
           WHERE id = ?`
        ).run(maxRow.max_seq + 1, setting.id)
      }
    }
  } catch {
    /* ignore */
  }

  db.exec(`
    CREATE INDEX IF NOT EXISTS idx_students_school_code ON students(school_id, student_code);
    CREATE UNIQUE INDEX IF NOT EXISTS idx_students_school_code_unique
      ON students(school_id, student_code)
      WHERE deleted_at IS NULL AND student_code IS NOT NULL;
  `)
}

export interface GenerateStudentCodeOptions {
  sequence?: number
  admissionYear?: number
  academicLevelId?: string | null
  sectionName?: string | null
  specializationAbbr?: string | null
}

export function generateStudentCode(
  db: Database.Database,
  schoolId: string,
  options?: GenerateStudentCodeOptions
): {
  student_code: string
  sequence_number: number
  pattern: string
} {
  // 1. Load the school's student_id_settings
  let settings = db
    .prepare(
      `SELECT * FROM student_id_settings WHERE school_id = ? AND deleted_at IS NULL LIMIT 1`
    )
    .get(schoolId) as StudentIdSettings | undefined

  if (!settings) {
    let initialSeq = 1
    try {
      const maxRow = db
        .prepare(
          `SELECT MAX(sequence_number) as max_seq FROM students WHERE school_id = ? AND deleted_at IS NULL`
        )
        .get(schoolId) as { max_seq: number | null } | undefined
      if (maxRow?.max_seq != null && maxRow.max_seq > 0) {
        initialSeq = maxRow.max_seq + 1
      }
    } catch {
      /* ignore */
    }

    settings = {
      id: '',
      school_id: schoolId,
      pattern: DEFAULT_STUDENT_ID_PATTERN,
      sequence_scope: 'school',
      next_sequence_number: initialSeq,
      created_at: new Date().toISOString(),
      updated_at: new Date().toISOString(),
      deleted_at: null,
      sync_status: 'pending',
      version: 1,
    }
  }

  // 2. Load the school's short_code
  const school = db
    .prepare(
      `SELECT id, name, short_code, school_code FROM schools WHERE id = ? AND deleted_at IS NULL LIMIT 1`
    )
    .get(schoolId) as School | undefined

  let schoolShortCode = school?.short_code?.trim() || ''
  if (!schoolShortCode) {
    schoolShortCode = school?.name ? normalizeShortCode(school.name) : 'SCH'
  }

  // 3. Read sequence number
  const sequence = options?.sequence ?? settings.next_sequence_number

  // 4. Resolve level short name if needed by custom patterns
  let levelShort = ''
  if (options?.academicLevelId) {
    try {
      const level = db
        .prepare(
          `SELECT name, short_name FROM academic_levels WHERE id = ? AND deleted_at IS NULL LIMIT 1`
        )
        .get(options.academicLevelId) as
        | { name: string; short_name: string | null }
        | undefined
      if (level) {
        levelShort =
          level.short_name?.trim() || deriveLevelShortName(level.name)
      }
    } catch {
      /* ignore */
    }
  }

  // 5. Generate the student code from the configured pattern
  const pattern = settings.pattern || DEFAULT_STUDENT_ID_PATTERN
  const student_code = buildStudentId(pattern, {
    SCHOOL: schoolShortCode,
    YEAR: options?.admissionYear ?? new Date().getFullYear(),
    LEVEL: levelShort,
    SECTION: options?.sectionName?.trim() || '',
    SPECIALIZATION: options?.specializationAbbr?.trim() || '',
    SEQ: padSequence(sequence, 4),
  })

  return {
    student_code,
    sequence_number: sequence,
    pattern,
  }
}

  create(input: CreateStudentInput): Student {
    const tx = this.db.transaction(() => {
      // 1. Resolve school_id and academic_level_id if omitted but class_id provided
      let schoolId = input.school_id ?? null
      let academicLevelId = input.academic_level_id ?? null
      if (input.class_id && (!schoolId || !academicLevelId)) {
        const cls = this.db
          .prepare(
            `SELECT school_id, academic_level_id FROM classes WHERE id = ? AND deleted_at IS NULL`
          )
          .get(input.class_id) as
          | { school_id: string; academic_level_id: string | null }
          | undefined
        if (cls) {
          if (!schoolId && cls.school_id) schoolId = cls.school_id
          if (!academicLevelId && cls.academic_level_id) {
            academicLevelId = cls.academic_level_id
          }
        }
      }

      const admissionYear =
        input.admission_year ?? new Date().getFullYear()

      let student_code: string | null = input.student_code?.trim() || null
      let sequence_number: number | null = input.sequence_number ?? null

      // 2. Generate permanent student_code atomically if not already supplied
      if (!student_code) {
        if (!schoolId) {
          throw new Error('school_id is required to generate student code.')
        }

        // Read student_id_settings
        let settings = this.db
          .prepare(
            `SELECT * FROM student_id_settings WHERE school_id = ? AND deleted_at IS NULL LIMIT 1`
          )
          .get(schoolId) as StudentIdSettings | undefined

        if (!settings) {
          const maxRow = this.db
            .prepare(
              `SELECT MAX(sequence_number) as max_seq FROM students WHERE school_id = ? AND deleted_at IS NULL`
            )
            .get(schoolId) as { max_seq: number | null } | undefined
          const initialSeq =
            maxRow?.max_seq != null && maxRow.max_seq > 0
              ? maxRow.max_seq + 1
              : 1

          const now = new Date().toISOString()
          settings = {
            id: randomUUID(),
            school_id: schoolId,
            pattern: DEFAULT_STUDENT_ID_PATTERN,
            sequence_scope: 'school',
            next_sequence_number: initialSeq,
            created_at: now,
            updated_at: now,
            deleted_at: null,
            sync_status: 'pending',
            version: 1,
          }
          this.db
            .prepare(
              `INSERT INTO student_id_settings (
                 id, school_id, pattern, sequence_scope, next_sequence_number,
                 created_at, updated_at, deleted_at, sync_status, version
               ) VALUES (
                 @id, @school_id, @pattern, @sequence_scope, @next_sequence_number,
                 @created_at, @updated_at, @deleted_at, @sync_status, @version
               )`
            )
            .run(settings)

          enqueueSync(
            this.db,
            'student_id_settings',
            settings.id,
            'INSERT',
            settings as unknown as Record<string, unknown>
          )
        }

        const sequence = settings.next_sequence_number

        const generated = generateStudentCode(this.db, schoolId, {
          sequence,
          admissionYear,
          academicLevelId,
          sectionName: input.section_name ?? null,
          specializationAbbr: input.specialization_abbr ?? null,
        })
        student_code = generated.student_code
        sequence_number = generated.sequence_number

        // Collision check
        const clash = this.db
          .prepare(
            `SELECT id FROM students WHERE school_id = ? AND student_code = ? AND deleted_at IS NULL LIMIT 1`
          )
          .get(schoolId, student_code) as { id: string } | undefined
        if (clash) {
          throw new Error(
            `Generated student ID “${student_code}” already exists. Adjust the pattern or sequence.`
          )
        }

        // Increment sequence in student_id_settings atomically
        const now = new Date().toISOString()
        const updatedSettings: StudentIdSettings = {
          ...settings,
          next_sequence_number: sequence + 1,
          updated_at: now,
          version: settings.version + 1,
          sync_status: 'pending',
        }
        this.db
          .prepare(
            `UPDATE student_id_settings
             SET next_sequence_number = @next_sequence_number,
                 updated_at = @updated_at,
                 version = @version,
                 sync_status = @sync_status
             WHERE id = @id`
          )
          .run(updatedSettings)

        enqueueSync(
          this.db,
          'student_id_settings',
          updatedSettings.id,
          'UPDATE',
          updatedSettings as unknown as Record<string, unknown>
        )
      } else if (schoolId) {
        // When student_code is explicitly provided, verify collision
        const clash = this.db
          .prepare(
            `SELECT id FROM students WHERE school_id = ? AND student_code = ? AND deleted_at IS NULL LIMIT 1`
          )
          .get(schoolId, student_code) as { id: string } | undefined
        if (clash) {
          throw new Error(`Student ID “${student_code}” already exists.`)
        }
      }

      // 3. Build student record with local UUID
      const base = this.buildBaseFields()
      const record: Student = {
        ...base,
        id: input.id || base.id,
        name: input.name.trim(),
        class_id: input.class_id,
        school_id: schoolId,
        student_code,
        admission_year: admissionYear,
        academic_level_id: academicLevelId,
        sequence_number,
        gender: input.gender ?? null,
        date_of_birth: input.date_of_birth ?? null,
        phone: input.phone ?? null,
        email: input.email ?? null,
        address: input.address ?? null,
        admission_type: input.admission_type ?? null,
        guardian_name: input.guardian_name ?? null,
        guardian_phone: input.guardian_phone ?? null,
        guardian_email: input.guardian_email ?? null,
        guardian_relationship: input.guardian_relationship ?? null,
        status: input.status ?? 'active',
        photo_path: input.photo_path ?? null,
        residence_type: input.residence_type === 'boarding' ? 'boarding' : 'day',
      }

      // 4. Insert student
      this.db
        .prepare(
          `INSERT INTO students (
             id, name, class_id, school_id, student_code,
             admission_year, academic_level_id, sequence_number,
             gender, date_of_birth, phone, email, address, admission_type,
             guardian_name, guardian_phone, guardian_email, guardian_relationship,
             status, photo_path, residence_type,
             created_at, updated_at, deleted_at, sync_status, version
           ) VALUES (
             @id, @name, @class_id, @school_id, @student_code,
             @admission_year, @academic_level_id, @sequence_number,
             @gender, @date_of_birth, @phone, @email, @address, @admission_type,
             @guardian_name, @guardian_phone, @guardian_email, @guardian_relationship,
             @status, @photo_path, @residence_type,
             @created_at, @updated_at, @deleted_at, @sync_status, @version
           )`
        )
        .run(record)

      // 5. Enqueue sync outbox entry
      enqueueSync(
        this.db,
        this.tableName,
        record.id,
        'INSERT',
        record as unknown as Record<string, unknown>
      )

      return record
    })

    return tx()
  }

Function: enqueueSync
Source: main/sync/queue.ts

import { randomUUID } from 'crypto'
import type Database from 'better-sqlite3'
import type { SyncOperation } from '../db/types'

/**
 * Enqueue a full-record snapshot for future sync.
 * MUST be called on every create / update / soft-delete.
 */
export function enqueueSync(
  db: Database.Database,
  tableName: string,
  recordId: string,
  operation: SyncOperation,
  payload: Record<string, unknown>
): void {
  // Payload must include id + updated_at (full record snapshot)
  const stmt = db.prepare(`
    INSERT INTO sync_queue (
      id, table_name, record_id, operation, payload, status, retry_count, last_attempt_at, created_at
    ) VALUES (
      @id, @table_name, @record_id, @operation, @payload, 'pending', 0, NULL, @created_at
    )
  `)

  stmt.run({
    id: randomUUID(),
    table_name: tableName,
    record_id: recordId,
    operation,
    payload: JSON.stringify(payload),
    created_at: new Date().toISOString(),
  })
}
